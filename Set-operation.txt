 36 void readset(bit a[],int n)
 37 {
 38   int i,k,x;
 39   if(n<0 || n>SIZE)
 40   {
 41     printf("Invalid size;must be between 0 and %d\n",SIZE);
 42     return;
 43   }
 44 printf("Enter %d elements:",n);
 45 for(i=0;i<n;i++)
 46 {
 47     if(scanf("%d",&x)!=1)return;
 48     for(k=0;k<SIZE;k++)
 49     {
 50         if(uset[k]==x)
 51         {
 52             a[k].x=1;break;
 53         }
 54     }
 55     if(k==SIZE)
 56         printf("%d is not in the universal set,ignored.\n",x);
 57 }

 73 void unionset(const bit a[],const bit b[],bit c[])

 78 void interset(const bit a[],const bit b[],bit c[])

 80     for (int k=0;k<SIZE;k++)
