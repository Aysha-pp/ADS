#include <stdio.h>
int largearray(int arr[],int n)
{
    int i;
    int m=arr[0];
    for(i=0;i<n;i++)
        if(arr[i]>m)
    {
        m=arr[i];  
    }
return m;
    
}
void main()
{
    int n,i ,arr[10];
    printf("Enter the elements:");
    scanf("%d",&n);
    printf("Enter the numbers: \n");
    for(i=0;i<n;i++)
        {
            scanf("%d",&arr[i]);
        }
    int m=largearray(arr,n);
    printf("Largest =  %d",m);
}
