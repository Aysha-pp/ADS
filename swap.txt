#include <stdio.h>
void swap(int *num1,int *num2)
{
int temp;

    temp=*num1;
    *num1=*num2;
    *num2=temp;
}
int main(){
int a, b;
printf("enter a number : \n");
scanf("%d",&a);
printf("enter a number : \n");
scanf("%d",&b);
printf("Before swapping:");
printf("a=%d\n",a);
printf("b=%d\n",b);
swap(&a,&b);
printf("after swapping:");
printf("a=%d\n",a);
printf("b=%d\n",b);
}
